# Yiddish Phonikud TTS — runtime dependencies.
#
# Every line below is imported by shipped code; nothing is listed "just in case".
# The phonikud-yi engine (notmax123/phonikud-yi-engine) and the BlueTTS 2.5
# bundle (notmax123/BlueTTS2.5-onnx) are NOT pip packages: both are downloaded
# at runtime with huggingface_hub.snapshot_download and need nothing beyond
# onnxruntime + numpy. torch, gradio, librosa and bluecodec are deliberately
# absent — the pointing model and all four BlueTTS graphs are ONNX exports, and
# the flow-matching loop, the 144->24 latent fold and the vocoder call are plain
# numpy in yiddish_phonikud/runtimes/blue_yi.py.

# --- web layer -------------------------------------------------------------
fastapi>=0.115            # app.py, api/routes_v1.py, api/routes_ui.py
# Imported directly, not only through fastapi: `starlette.exceptions` in app.py
# and `starlette.concurrency.run_in_threadpool` in both routers. fastapi pins it
# transitively today, but a future fastapi that loosened or swapped that pin
# would break the error envelope and both threadpool call sites with no
# requirements-level signal, so it is declared.
starlette>=0.37
uvicorn[standard]>=0.30   # app.py __main__ and the Dockerfile CMD
python-multipart>=0.0.9   # POST /generate reads Form() fields from a FormData
jinja2>=3.1               # routes_ui.py Jinja2Templates(templates/)
pydantic>=2               # api/dto.py — every request/response model

# --- engine + runtimes -----------------------------------------------------
# snapshot_download: the ~1.23 GB engine and the ~282 MB blue-yi bundle.
# [oauth] pulls authlib + itsdangerous for attach_huggingface_oauth (ABE101 editor).
huggingface_hub[oauth]>=0.25
# Tensor plumbing for the runtime, plus audio.pcm16_wav's vectorised packing.
numpy>=1.26
# The v5 pointing model and blue-yi's four graphs.
onnxruntime>=1.18
